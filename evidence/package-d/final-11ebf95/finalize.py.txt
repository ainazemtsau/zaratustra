"""Run only after both exact-source gates complete; all counts are observed."""
import hashlib, importlib, json, re, shutil, subprocess, sys
from pathlib import Path

root=Path.cwd();proof=root/'evidence/package-d/final-11ebf95'
sys.path.insert(0,str(root))
pin='11ebf956bafc2bbca4a3685e37d117474f11c151'
assert subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()==pin
assert not subprocess.check_output(['git','diff',pin,'--','packages','tests','tools','src','pyproject.toml','uv.lock'])
native=json.loads((proof/'native-11ebf95/exit.json').read_bytes())
scoped=json.loads((proof/'scoped-11ebf95/exit.json').read_bytes())
assert native['source']==scoped['source']==pin
assert native['returncode']==scoped['returncode']==0
native_text=(proof/'native-11ebf95/stdout.bin').read_text(encoding='utf-8')
scoped_text=(proof/'scoped-11ebf95/stdout.bin').read_text(encoding='utf-8')
assert 'PASS: native build, hygiene, types, boundaries, tests, report structure' in native_text
native_count=int(re.search(r'(\d+) passed in ',native_text).group(1))
scoped_count=int(re.search(r'(\d+) passed in ',scoped_text).group(1))
expected={}
for package,module in [('A','test_negative_controls'),('B','test_continuation_controls'),('C','test_sleep_controls'),('D','test_delivery_controls')]:
    seeds=importlib.import_module('tests.restart.'+module).SEEDS
    for seed in seeds:
        assert seed[0] not in expected
        expected[seed[0]]=package
controls=[]
for path in sorted((proof/'scoped-11ebf95/negatives').glob('*.json')):
    result=json.loads(path.read_bytes())
    assert result['returncode']==1 and result['injection_reached'] and result['behavioral_failure'],path
    controls.append({'seed':result['seed'],'package':expected[result['seed']],
                     'path':path.relative_to(proof).as_posix(),'test':result['test'],
                     'reached':True,'specific_behavioral_failure':True})
assert {r['seed'] for r in controls}==set(expected)
groups={package:sum(r['package']==package for r in controls) for package in 'ABCD'}
(proof/'CONTROL-MANIFEST.json').write_text(json.dumps({'count':len(controls),'by_package':groups,'controls':controls},indent=2),encoding='utf-8')
physical=proof/'native-11ebf95/physical-builds';physical.mkdir(exist_ok=False)
builds=[]
for path in sorted((proof/'install-11ebf95/physical-builds').iterdir()):
    if not path.name.endswith(('.whl','.tar.gz')):continue
    observed=(root/'dist'/path.name).read_bytes()
    assert observed==path.read_bytes(),path
    shutil.copyfile(root/'dist'/path.name,physical/path.name)
    builds.append({'file':path.name,'sha256':hashlib.sha256(observed).hexdigest(),'bytes':len(observed)})
assert len(builds)==4
(proof/'BUILD-MANIFEST.json').write_text(json.dumps({'source_commit':pin,'distributions':builds,
 'native_and_outside_build_bytes_equal':True,'source_wheel_installed_mapping':'install-11ebf95/manifest.json',
 'dependency_bytes':'DEPENDENCY-MANIFEST.json','owner_bundle':'OWNER-BUNDLE-MANIFEST.json',
 'outside_git_source_and_runtime':True,'actual_installed_console':True,
 'ordinary_trace':'Declared fictional OAuth/upstream; no semantic G14 claim.'},indent=2),encoding='utf-8')
provenance={'source_commit':pin,'accepted_c_candidate':'7d211480d1dd5067bf52c3a654cc1a713368310d',
 'native':native,'native_tests':native_count,'scoped':scoped,'scoped_tests':scoped_count,
 'reached_behavioral_controls':len(controls),'controls_by_package':groups,
 'contract':'wire/v1','conversation_method':'conversation/1','backup':'backup/v1','diagnostics':'diagnostics/v1',
 'runtime':'Python3.13.7/nativeSQLite3.50.4/Pi1.3.0/Node22.19.0; declared legacy full gate only SQLiteDLL3.53.3',
 'live_g14':'Pending separate fresh physical reviewer; this writer made no real model call. Root separately made one bounded Luna auth preflight send (10input/5output), not G14. Independent installed1b6 G14 made1send HTTP400 unsupported configuration_update before useful response/correction/reopen; usage unknown. Corrected-source live G14 remains separate.',
 'owner_acceptance':'D and first release pending','next':'solmax',
 'candidate_policy':'Evidence/docs/RESULT only after this source pin. Candidate commit is returned separately.',
 'prior_runs':'Original source095/786/627/20d/f072/1b6 failures and incomplete runs retained/indexed, including1b6 SURVIVED checksum control and failed live G14. F072 scoped280PASS does not close independent F-D-01.',
 'windows_publication':'Original holder unknown. Reproduced file-sharing class mitigated with finite2s retry Windows5/32/33; permanent/unrelated failures remain errors.',
 'credentials':'No real credential/token values exported. Disposable FICTIONAL SQLite contains random test credential hashes only.'}
(proof/'PROVENANCE.json').write_text(json.dumps(provenance,indent=2),encoding='utf-8')
p=root/'docs/restart/PACKAGE-D-RETURN.md';body=p.read_text(encoding='utf-8')
body=body.replace('Current full native/scoped completion and independent review are pending.',
 f'Exact-source full native gate passed {native_count} tests plus build, hygiene, types,\nboundaries and report structure. p.write_text(body,encoding='utf-8')
