"""Run only after both exact final-source gates actually exit successfully."""
from pathlib import Path
import hashlib, importlib, json, re, shutil, subprocess, sys

r = Path.cwd()
p = r / 'evidence/package-d/final-f80180a'
pin = 'f80180a950ce7a3e21d30c86e610ae4a935320e3'
assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip() == pin
assert not subprocess.check_output(['git', 'diff', pin, '--', 'packages', 'src', 'tests', 'tools', 'pyproject.toml', 'uv.lock'])
exits = {}
counts = {}
for kind in ['native', 'scoped']:
    d = p / (kind + '-f80180a')
    receipt = json.loads((d / 'exit.json').read_text(encoding='utf-8'))
    assert receipt['source'] == pin and receipt['returncode'] == 0, receipt
    text = (d / 'stdout.bin').read_text(encoding='utf-8')
    counts[kind] = int(re.search(r'(\d+) passed in ', text).group(1))
    if kind == 'native':
        assert 'PASS: native build, hygiene, types, boundaries, tests, report structure' in text
    exits[kind] = receipt
sys.path.insert(0, str(r))
expected = {}
for package, module in [('A', 'test_negative_controls'), ('B', 'test_continuation_controls'), ('C', 'test_sleep_controls'), ('D', 'test_delivery_controls')]:
    for seed in importlib.import_module('tests.restart.' + module).SEEDS:
        assert seed[0] not in expected
        expected[seed[0]] = package
controls = []
for f in sorted((p / 'scoped-f80180a/negatives').glob('*.json')):
    item = json.loads(f.read_text(encoding='utf-8'))
    assert item['returncode'] == 1 and item['injection_reached'] and item['behavioral_failure'], f
    controls.append({'seed': item['seed'], 'package': expected[item['seed']], 'test': item['test'], 'path': f.relative_to(p).as_posix(), 'sha256': hashlib.sha256(f.read_bytes()).hexdigest(), 'reached': True, 'specific_behavioral_failure': True})
assert {c['seed'] for c in controls} == set(expected)
groups = {k: sum(c['package'] == k for c in controls) for k in 'ABCD'}
(p / 'CONTROL-MANIFEST.json').write_text(json.dumps({'count': len(controls), 'by_package': groups, 'controls': controls}, indent=2), encoding='utf-8')
physical = p / 'native-f80180a/physical-builds'
physical.mkdir(exist_ok=False)
builds = []
for f in sorted((p / 'install-f80180a/physical-builds').iterdir()):
    if not f.name.endswith(('.whl', '.tar.gz')):
        continue
    actual = (r / 'dist' / f.name).read_bytes()
    assert actual == f.read_bytes(), f
    shutil.copyfile(r / 'dist' / f.name, physical / f.name)
    builds.append({'file': f.name, 'sha256': hashlib.sha256(actual).hexdigest(), 'bytes': len(actual)})
assert len(builds) == 4
(p / 'BUILD-MANIFEST.json').write_text(json.dumps({'source_commit': pin, 'distributions': builds, 'native_and_outside_build_bytes_equal': True, 'source_wheel_installed_files': 20, 'dependency_installed_files': 145, 'installation': 'OUTSIDE-BUILD-INSTALL-MANIFEST.json', 'owner_archive': 'OWNER-BUNDLE-MANIFEST.json', 'real_owner_entry_mechanics': 'OWNER-START-CONTROLLED.json'}, indent=2), encoding='utf-8')
summary = {'source_commit': pin, 'accepted_c': '7d211480d1dd5067bf52c3a654cc1a713368310d', 'exits': exits, 'test_counts': counts, 'reached_controls': len(controls), 'controls_by_package': groups, 'production_files': 20, 'dependency_files': 145, 'core_contract': 'wire/v1', 'methods': ['conversation-routing/1', 'conversation/2'], 'runtime': 'Python3.13.7/SQLite3.50.4/Pi1.3.0/Node22.19.0; legacy full gate only declared SQLite3.53.3 DLL', 'native_resource': {'mode': 'calls-only', 'finite_shared_call_reservation': True, 'hard_output_token_or_money_maximum': False, 'unsupported_field_omitted': True, 'owner_quote': 'Да, ограничиваем число обращений — продолжай (рекомендуется)', 'prospective_g14_chain_limit': 8, 'prospective_g14_total_upper': 24}, 'writer_live_calls': 0, 'prior_g14': {'source': '83af4d4a529ca73d06d566db62a3586f2a432af9', 'verdict': 'NOT_READY/F-D-08', 'requests': 4, 'tokens': 21790, 'prior_series': '3/3 exhausted, immutable', 'requested4096': 'not verified enforcement'}, 'review_and_new_g14': 'separate physical reviewer/lead-owned evidence; never inferred from deterministic PASS', 'owner_acceptance': 'D and first release pending; resource-choice acceptance only', 'negative_history': 'All original failures, INFRA_STOP and incomplete/superseded runs retained; no status substitution', 'next': 'solmax'}
summary['current_g14'] = json.loads((p / 'INDEPENDENT-G14-OUTCOME-ROOT-REPORT.json').read_text(encoding='utf-8'))
summary['review_and_new_g14'] = 'G14 PASS_LIMITED_SINGLE_TRACE; formal independent source readiness receipt pending, never owner acceptance'
(p / 'PROVENANCE.json').write_text(json.dumps(summary, indent=2), encoding='utf-8')
print(json.dumps(summary))
