def sha(data):
    return hashlib.sha256(data).hexdigest()


def write(name, value):
    (E / name).write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n', encoding='utf-8')


def read(path):
    return json.loads((E / path).read_text(encoding='utf-8'))


def entry(path, relative_to=E):
    data = path.read_bytes()
    return {'path': path.relative_to(relative_to).as_posix(), 'sha256': sha(data), 'bytes': len(data)}

def count_passes(path):
    text = path.read_text(encoding='utf-8', errors='replace')
    matches = re.findall(r'(\d+) passed in ([0-9.]+)s', text)
    assert matches, path
    return {'tests_passed': int(matches[-1][0]), 'pytest_seconds': float(matches[-1][1])}
