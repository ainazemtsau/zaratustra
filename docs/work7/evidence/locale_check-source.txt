from pathlib import Path

base=Path(sys.argv[1]);label=sys.argv[2]
script='''import sys

    return authorize_local(prompt,channel="local-chat",actor="executor-fictional-adapter",source_ref="Work7:simulated-prior-permission")

'''
args=[str(base/'venv/Scripts/python.exe'),'-I','-c',script,str(base/'workspace'),(base/'receipt-query.json').read_text(encoding='utf-8')]

(base/f'{label}.json').write_text(json.dumps(dict(command=args,cwd=str(base/'unrelated-cwd'),exit=r.returncode,stdout=r.stdout.decode('ascii'),stderr=r.stderr.decode('ascii'),encoding='ascii'),indent=2),encoding='utf-8')
