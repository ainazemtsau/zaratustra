from pathlib import Path
import hashlib, json, re, shutil, subprocess, zipfile

root=Path(__file__).resolve().parents[1]
docs=root/'docs/work7'; ev=docs/'evidence'
def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def git(*args): return subprocess.check_output(['git',*args],cwd=root)
gate=json.loads((ev/'native-deliver.json').read_text(encoding='utf-8'))
assert gate['exit']==0
assert '140 passed in 29.52s' in (ev/'native-deliver.stdout').read_text(encoding='utf-8')

assert git('diff',build['base'],build['candidate'])==(ev/'implementation.patch').read_bytes()
manifest=json.loads((ev/'retained-trial-manifest.json').read_text(encoding='utf-8'))
assert sha(ev/'retained-trial.zip')==manifest['archive_sha256']
with zipfile.ZipFile(ev/'retained-trial.zip') as archive:
    for name,digest in manifest['files'].items():
        assert hashlib.sha256(archive.read(name)).hexdigest()==digest,name
    assert all(archive.getinfo(name).is_dir() for name in manifest['directories'])
old_paths=['validation.config',*[f'docs/work{i}' for i in range(1,7)],*[f'src/zaratustra/core/migration_v{i}.py' for i in range(1,6)]]
assert not git('diff',build['base'],'--name-only','--',*old_paths).strip()
assert git('diff',build['base'],'--name-only','--','tests').decode().strip()=='tests/zaratustra/core/test_results.py'
assert not git('diff',build['candidate'],'--name-only','--','src','tests','tools','pyproject.toml','uv.lock').strip()
report=(root/'RESULT.md').read_text(encoding='utf-8')
old='Final full --deliver outcome and exact count: docs/work7/DELIVERY.md and\nnative-deliver.json/stdout/stderr.'
new='Final full --deliver PASS: 140 tests in 29.52s, build, hygiene, ruff, mypy,\ntwo import boundaries and report structure. Raw: native-deliver.json/stdout/stderr;\nexact command and duration: docs/work7/DELIVERY.md.'
