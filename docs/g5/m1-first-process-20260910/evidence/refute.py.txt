def guard():
    for name in ('STOP', 'STEER.md'):
        if (ROOT / name).exists():
            raise RuntimeError('Read and resolve ' + name)

def dump(path, value):
    path.write_text(json.dumps(value, indent=2, default=str)+'\n', encoding='utf-8')

def check(name, value, detail=None):
    checks.append({'name': name, 'pass': bool(value), 'detail': detail})
    dump(BASE / 'checks.json', checks)
    if not value:
        raise AssertionError(name)

def footprint(path):
    return {p.relative_to(path).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest()
            if p.is_file() else None for p in sorted(path.rglob('*'))}

def confirm(path, request):
    caller = authorize_local(prepare_authorization(path, request), channel='local-chat',
                             actor='independent-G5-reviewer', source_ref=RUN)
    return caller

def work(path, identity=None):
    return next(r for r in read_records(path).records if isinstance(r, Work)
                and (identity is None or r.id == identity))

def context_query(path, identity, **options):
    state = read_records(path)
    return ContextQuery.model_validate(dict(workspace_id=state.workspace_id, work_id=identity,
        process_id=work(path, identity).process_id, expected_revision=state.state_revision,
        max_bytes=1048576) | options)

def metadata_query(path, identity, visible, selected=None, **options):
    state = read_records(path)
    return ProcessQuery.model_validate(dict(workspace_id=state.workspace_id,
        process_id=work(path, identity).process_id, work_id=identity,
        expected_revision=state.state_revision, visible_work_ids=visible,
        selected_work_id=selected, max_bytes=1048576) | options)

class Host:
    def __init__(self, name, restored=None):
        self.base = BASE / name
        self.base.mkdir()
        self.path = restored or self.base / 'workspace'
        self.seq = 0

    def request(self, identity, operation, **payload):
        state = read_records(self.path)
        fields = dict(operation=operation, operation_id=uuid4(), workspace_id=state.workspace_id,
                      work_id=identity, expected_revision=state.state_revision, provenance=RUN)
        if operation in ('authorize_artifact', 'publish_artifact'):
            artifact = next(r for r in state.records if isinstance(r, Artifact) and r.work_id == identity)
            fields.update(version=2, artifact_id=artifact.id, artifact_revision=artifact.revision)
        return MutationRequest.model_validate(fields | payload)

    def execute(self, request, content=None):
        guard()
        self.seq += 1
        prefix = self.base / f'{self.seq:03d}-{request.operation}'
        caller = confirm(self.path, request)
        dump(prefix.with_suffix('.request.json'), request.model_dump(mode='json'))
        dump(prefix.with_suffix('.authority.json'), caller.confirmation.model_dump(mode='json'))
        dump(prefix.with_suffix('.before.json'), footprint(self.path))
        if content is not None:
            prefix.with_suffix('.input.json').write_bytes(content)
        receipt = apply_mutation(self.path, request, caller, content=content)
        dump(prefix.with_suffix('.receipt.json'), receipt.model_dump(mode='json'))
        dump(prefix.with_suffix('.after.json'), footprint(self.path))
        return receipt

    def bootstrap(self):
        guard()
        self.path.mkdir()
        init_workspace(self.path)
        migrate_workspace(self.path, target_version=7)
        initial = initial_records()
        dump(self.base / 'bootstrap.json', initial.model_dump(mode='json'))
        create_initial_records(self.path, initial)
        identity = work(self.path).id
        self.execute(self.request(identity, 'authorize_work'))
        self.execute(self.request(identity, 'set_work_requirements', requirements=initial_requirements()))
        state = read_records(self.path)
        self.execute(binding_request(REGISTRY, reference(), operation_id=uuid4(),
            workspace_id=state.workspace_id, work_id=identity, expected_revision=state.state_revision,
            provenance=RUN))
        return identity

    def accept(self, identity, raw):
        self.execute(self.request(identity, 'authorize_artifact'))
        publication = self.request(identity, 'publish_artifact',
            content_sha256=hashlib.sha256(raw).hexdigest(), content_size=len(raw))
        self.execute(publication, raw)
        ref = ArtifactReference(artifact_id=publication.artifact_id, version_id=publication.operation_id,
                                sha256=hashlib.sha256(raw).hexdigest())
        state = read_records(self.path)
        handoff = Handoff(kind='handoff', version=1, handoff_id=uuid4(),
            workspace_id=state.workspace_id, process=work(self.path, identity).process_id,
            related_work=identity, intent='accepted_result', source_revision=state.state_revision,
            result=ref, provenance=RUN, created_by='G5 fictional host')
        self.execute(handoff_request(handoff.model_dump_json().encode(), source_ref=RUN))
        return ref

    def propose(self, identity, name, registry=REGISTRY):
        query = context_query(self.path, identity)
        caller = confirm(self.path, query)
        dump(self.base / (name+'.query.json'), query.model_dump(mode='json'))
        dump(self.base / (name+'.authority.json'), caller.confirmation.model_dump(mode='json'))
        before = footprint(self.path)
        result = propose_result(self.path, query, caller, registry, operation_id=uuid4(),
                                next_work_id=uuid4(), next_artifact_id=uuid4())
        check(name+' proposal has no writes', before == footprint(self.path))
        dump(self.base / (name+'.proposal.json'), result.model_dump(mode='json'))
        return result

    def read(self, name, query, caller='auto', registry=REGISTRY, **options):
        if caller == 'auto':
            caller = confirm(self.path, query)
        dump(self.base / (name+'.query.json'), query.model_dump(mode='json'))
        if caller is not None:
            dump(self.base / (name+'.authority.json'), caller.confirmation.model_dump(mode='json'))
        before = footprint(self.path)
        response = read_capabilities(self.path, query, caller, registry, **options)
        (self.base / (name+'.response.json')).write_bytes(response.output)
        check(name+' read unchanged', before == footprint(self.path))
        value = json.loads(response.output)
        check(name+' seven answers', set(value['answers']) == NAMES)
        return value

def refusal(host, name, action, code):
    before = footprint(host.path)
    dump(host.base / (name+'.before.json'), before)
    try:
        action()
    except (MutationError, PackError) as error:
        dump(host.base / (name+'.refusal.json'), {'code': error.code, 'detail': str(error)})
        check(name+' expected refusal', error.code == code, {'actual': error.code, 'expected': code})
    else:
        check(name+' expected refusal', False)
    after = footprint(host.path)
    dump(host.base / (name+'.after.json'), after)
    check(name+' unchanged', before == after)

def snapshot(host, name):
    manifest = retain_trial(host.path, host.base / (name+'.zip'))
    dump(host.base / (name+'-manifest.json'), manifest)
    return manifest
