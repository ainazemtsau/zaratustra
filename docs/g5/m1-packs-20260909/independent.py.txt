def save(name, value):
    target = OUT / name
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(json.dumps(value, indent=2, ensure_ascii=False, default=str) + "\n", encoding="utf-8", newline="\n")

def check(name, condition, **details):
    row = dict(case=name, verdict="PASS" if condition else "FAIL", **details)
    LOG.append(row)
    save("checks.json", LOG)
    print(json.dumps(row, ensure_ascii=True), flush=True)
    if not condition:
        raise AssertionError(name)

def hash_bytes(data):
    return hashlib.sha256(data).hexdigest()

def inventory(path):
    return dict(files={p.relative_to(path).as_posix(): hash_bytes(p.read_bytes()) for p in sorted(path.rglob("*")) if p.is_file()}, directories=sorted(p.relative_to(path).as_posix() + "/" for p in path.rglob("*") if p.is_dir()))

def db_rows(path):
    db = read_workspace(path).database
    with sqlite3.connect(db.as_uri() + "?mode=ro", uri=True) as connection:
        names = [r[0] for r in connection.execute("SELECT name FROM sqlite_master WHERE type='table' ORDER BY name")]
        return {name: connection.execute('SELECT * FROM "' + name + '" ORDER BY 1').fetchall() for name in names}

def confirm(path, value):
    return authorize_local(prepare_authorization(path, value), channel="local-chat", actor="fresh-physical-G5-fictional-review", source_ref=f"Owner authorized G5 in source task 01a0865e-b759-7bf3-9d21-e86819e52156; review {TASK}; fictional fixture only; owner acceptance pending")

def work(path, identity=None):
    return next(r for r in read_records(path).records if isinstance(r, Work) and (identity is None or r.id == identity))

def request(path, identity, operation, **payload):
    state = read_records(path)
    values = dict(operation_id=uuid4(), workspace_id=state.workspace_id, work_id=identity, operation=operation, expected_revision=state.state_revision, provenance="Fresh G5 fictional bounded request")
    if operation in ("authorize_artifact", "publish_artifact", "restore_artifact"):
        artifact = next(r for r in state.records if isinstance(r, Artifact) and r.work_id == identity)
        values.update(version=2, artifact_id=artifact.id, artifact_revision=artifact.revision)
    return MutationRequest.model_validate(values | payload)

SEQUENCE = 0
def execute(path, value, content=None):
    global SEQUENCE
    SEQUENCE += 1
    auth = confirm(path, value)
    stem = f"operations/{SEQUENCE:03d}-{value.operation}"
    save(stem + "-input.json", dict(path=path, request=value.model_dump(mode="json"), confirmation=auth.confirmation.model_dump(mode="json"), content_hex=content.hex() if content is not None else None))
    receipt = apply_mutation(path, value, auth, content=content)
    save(stem + "-receipt.json", receipt.model_dump(mode="json"))
    return receipt

def retain(path, name):
    manifest = retain_trial(path, OUT / (name + ".zip"))
    save(name + "-manifest.json", manifest)
    return OUT / (name + ".zip"), manifest

def clone(backup, name):
    return restore_snapshot(backup[0], OUT / name, backup[1])

def refused(name, fn, *, path=None, codes=()):
    before = inventory(path) if path else None
    try:
        fn()
    except Exception as error:
        code = getattr(error, "code", type(error).__name__)
        check(name, (not codes or code in codes) and (path is None or inventory(path) == before), code=code, detail=str(error), unchanged_layout_and_bytes=path is None or inventory(path) == before, before=before)
        return
    check(name, False, reason="Expected refusal but call returned")

def ref(**changes):
    return PackReference.model_validate(dict(pack_id="review.unseen", pack_version="3.1.4", process_type="review.unseen-type", contract_version=1, state_version=1) | changes)

class Rule:
    def __init__(self, label):
        self.label = label
        self.calls = []

    def next_work(self, current, accepted, work_id, artifact_id):
        self.calls.append(dict(work=str(current.id), binding=current.pack_binding.model_dump(mode="json"), accepted=accepted.hex()))
        return NextWork(work_id=work_id, artifact_id=artifact_id, goal=f"Fictional continuation {self.label}", expected_result="Independent retained observation", acceptance=("Fictional check",), boundaries=("Review-only fixture",), budget="One review transition", executor_requirements=("review.external-rule",), artifact_title="Independent next artifact", authority_scope="work_metadata")

def installed(reference, rule=None):
    return PackRegistry((PackRegistration(reference, rule or Rule("unused")),))

def bind(path, identity, reference, registry):
    state = read_records(path)
    return binding_request(registry, reference, operation_id=uuid4(), workspace_id=state.workspace_id, work_id=identity, expected_revision=state.state_revision, provenance="Fresh G5 explicit fictional binding")

def query(path, identity):
    state = read_records(path)
    return ContextQuery(workspace_id=state.workspace_id, work_id=identity, process_id=work(path, identity).process_id, expected_revision=state.state_revision, max_bytes=1048576)

def proposal(path, identity, registry, q=None, caller="new"):
    q = q or query(path, identity)
    auth = confirm(path, q) if caller == "new" else caller
    return propose_result(path, q, auth, registry, operation_id=uuid4(), next_work_id=uuid4(), next_artifact_id=uuid4())

def observe(path, identity):
    data = b"Fresh G5 observation: this is not the T1 batch JSON rule."
    execute(path, request(path, identity, "authorize_artifact"))
    publication = request(path, identity, "publish_artifact", content_sha256=hash_bytes(data), content_size=len(data))
    execute(path, publication, data)
    h = Handoff(kind="handoff", version=1, handoff_id=uuid4(), workspace_id=publication.workspace_id, process=work(path, identity).process_id, related_work=identity, intent="accepted_result", source_revision=read_records(path).state_revision, result=ArtifactReference(artifact_id=publication.artifact_id, version_id=publication.operation_id, sha256=hash_bytes(data)), provenance="Independent fictional evidence", created_by="Fresh G5 review")
    execute(path, handoff_request(h.model_dump_json().encode(), source_ref=TASK))
    return h.result, data
