def digest(data): return hashlib.sha256(data).hexdigest()
def save(name,value):
    p=OUT/name
    p.parent.mkdir(parents=True,exist_ok=True)
    p.write_text(json.dumps(value,indent=2,ensure_ascii=False,default=str)+"\n",encoding="utf-8",newline="\n")
def check(name,condition,**details):
    row=dict(case=name,verdict="PASS" if condition else "FAIL",**details)
    LOG.append(row)
    save("checks.json",LOG)
    print(json.dumps(row,ensure_ascii=True),flush=True)
    if not condition: raise AssertionError(name)
def git(*args): return subprocess.check_output(["git",*args],cwd=ROOT)
def j(path): return json.loads(path.read_bytes())
def rows(path):
    with sqlite3.connect(read_workspace(path).database.as_uri()+"?mode=ro",uri=True) as con:
        tables=[r[0] for r in con.execute("SELECT name FROM sqlite_master WHERE type='table' ORDER BY name")]
        return {t:con.execute('SELECT * FROM "'+t+'" ORDER BY 1').fetchall() for t in tables}
def confirm(path,value):

def snapshot(path,name):

def checked_zip(archive,manifest):
    with zipfile.ZipFile(archive) as z:
        names=z.namelist()
        files={i.filename:digest(z.read(i)) for i in z.infolist() if not i.is_dir()}
        dirs=sorted(i.filename for i in z.infolist() if i.is_dir())
        check("archive exact CRC layout "+archive.name,z.testzip() is None and len(set(names))==len(names) and files==manifest["files"] and dirs==manifest["directories"] and digest(archive.read_bytes())==manifest["archive_sha256"],archive_sha256=digest(archive.read_bytes()),files=len(files),directories=len(dirs))
def wheel_check(path,commit):
    with zipfile.ZipFile(path) as z:
        members=z.namelist()
        mismatch=[]
        source_members=[]
        for name in members:
            if name.startswith("zaratustra/") and name.endswith(".py"):
                source_members.append("src/"+name)
                if z.read(name)!=git("show",commit+":src/"+name): mismatch.append(name)
        expected=[n for n in git("ls-tree","-r","--name-only",commit,"--","src/zaratustra").decode().splitlines() if n.endswith(".py")]
        record=next(n for n in members if n.endswith(".dist-info/RECORD"))
        records=list(csv.reader(io.StringIO(z.read(record).decode())))
        invalid_record=[]
        for name, hashed, size in records:
            if not hashed: continue
            data=z.read(name)
            actual="sha256="+base64.urlsafe_b64encode(hashlib.sha256(data).digest()).decode().rstrip("=")
            if actual!=hashed or int(size)!=len(data): invalid_record.append(name)
        check("wheel source identity "+path.name,not mismatch and sorted(source_members)==sorted(expected) and not invalid_record and z.testzip() is None,source=commit,python_files=len(expected),sha256=digest(path.read_bytes()),mismatches=mismatch,invalid_RECORD=invalid_record)

    actual={p.relative_to(E).as_posix() for p in E.rglob("*") if p.is_file()}-{"manifest.json"}

        data=git("show",SOURCE+":"+name)
