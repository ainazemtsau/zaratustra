def footprint(path):
    return {p.relative_to(path).as_posix(): digest(p.read_bytes()) if p.is_file() else None
            for p in sorted(path.rglob('*'))}

class Reader:
    def __init__(self, refs=(), variant=None, hook=None):
        self.refs, self.variant, self.hook = refs, variant, hook
        self.calls = 0
        self.inputs = []
    def describe(self, m):
        self.calls += 1
        self.inputs.append(m.model_dump(mode='json'))
        active = [w for w in m.works if w.status not in ('done','cancelled')]
        available = tuple(w.id for w in active if w.status == 'ready'
                          and 'g5:choice' not in w.executor_requirements)
        blocked = tuple(Notice(key=str(w.id), work_id=w.id, text='Fictional choice or readiness')
                        for w in active if w.id not in available)
        result = CapabilitySelection(current_status=f'Visible active {len(active)}',
            items_needing_attention=blocked, open_decisions=blocked,
            available_works=available, blocked_works=blocked,
            recent_important_results=tuple(r.id for r in reversed(m.results)),
            context_requirements=ContextRequirements(notes=('Exact selected grants only',),
                                                     references=self.refs))
        if self.hook:
            self.hook(m)
        if self.variant:
            return self.variant(result, m)
        return result

    save(f/'history-before.json',read_history(path).model_dump(mode='json'))
    out=read_capabilities(path,q,a,reg if reg is not None else registry(path,rd),
